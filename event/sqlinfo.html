3. Conoha WING (phpMyAdmin) 실행용 SQL 스크립트

코노하윙 phpMyAdmin [SQL] 탭에 그대로 복사해서 실행하시면 됩니다.

-- ========================================================
-- Semo QR Event System Database DDL Script
-- Target DBMS: MariaDB / MySQL (Conoha WING Server)
-- UTF-8 (utf8mb4) Encoding Supported
-- ========================================================

-- 1. 언어 관리 테이블
CREATE TABLE IF NOT EXISTS `languages` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `name` VARCHAR(50) NOT NULL,
    `code` VARCHAR(10) NOT NULL UNIQUE,
    `is_active` TINYINT(1) DEFAULT 1,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2. 언어 콘텐츠 테이블
CREATE TABLE IF NOT EXISTS `language_contents` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `language_id` INT NOT NULL,
    `text` VARCHAR(200) NOT NULL,
    `reading` VARCHAR(200) NULL,
    `meaning` VARCHAR(200) NOT NULL,
    `explanation` TEXT NULL,
    `example_text` VARCHAR(300) NULL,
    `example_meaning` VARCHAR(300) NULL,
    `image_url` VARCHAR(500) NULL,
    `level` INT DEFAULT 1,
    `is_active` TINYINT(1) DEFAULT 1,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT `fk_contents_language` FOREIGN KEY (`language_id`) REFERENCES `languages` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 3. 지역 관리 테이블
CREATE TABLE IF NOT EXISTS `regions` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `name` VARCHAR(50) NOT NULL,
    `is_active` TINYINT(1) DEFAULT 1
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 4. 제휴 매장 테이블 (Phase 2)
CREATE TABLE IF NOT EXISTS `stores` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `region_id` INT NOT NULL,
    `store_name` VARCHAR(100) NOT NULL,
    `store_type` VARCHAR(50) NULL,
    `address` VARCHAR(200) NULL,
    `manager_name` VARCHAR(50) NULL,
    `is_active` TINYINT(1) DEFAULT 1,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT `fk_stores_region` FOREIGN KEY (`region_id`) REFERENCES `regions` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 5. 당첨 경품 테이블
CREATE TABLE IF NOT EXISTS `event_prizes` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `name` VARCHAR(100) NOT NULL,
    `semo_point` INT DEFAULT 0,
    `probability` INT DEFAULT 0,
    `stock` INT DEFAULT -1,
    `is_active` TINYINT(1) DEFAULT 1
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 6. QR 코드 관리 테이블
CREATE TABLE IF NOT EXISTS `event_qr` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `qr_code_key` VARCHAR(50) NOT NULL UNIQUE,
    `content_id` INT NOT NULL,
    `region_id` INT NOT NULL,
    `store_id` INT NULL,
    `scan_count` INT DEFAULT 0,
    `is_active` TINYINT(1) DEFAULT 1,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT `fk_qr_content` FOREIGN KEY (`content_id`) REFERENCES `language_contents` (`id`) ON DELETE CASCADE,
    CONSTRAINT `fk_qr_region` FOREIGN KEY (`region_id`) REFERENCES `regions` (`id`) ON DELETE CASCADE,
    CONSTRAINT `fk_qr_store` FOREIGN KEY (`store_id`) REFERENCES `stores` (`id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 7. 스캔/참여 이력 로그 테이블
CREATE TABLE IF NOT EXISTS `event_scan_history` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `qr_id` INT NOT NULL,
    `content_id` INT NOT NULL,
    `region_id` INT NOT NULL,
    `store_id` INT NULL,
    `prize_id` INT NULL,
    `device_key` VARCHAR(200) NULL,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT `fk_history_qr` FOREIGN KEY (`qr_id`) REFERENCES `event_qr` (`id`) ON DELETE CASCADE,
    CONSTRAINT `fk_history_content` FOREIGN KEY (`content_id`) REFERENCES `language_contents` (`id`) ON DELETE CASCADE,
    CONSTRAINT `fk_history_region` FOREIGN KEY (`region_id`) REFERENCES `regions` (`id`) ON DELETE CASCADE,
    CONSTRAINT `fk_history_store` FOREIGN KEY (`store_id`) REFERENCES `stores` (`id`) ON DELETE SET NULL,
    CONSTRAINT `fk_history_prize` FOREIGN KEY (`prize_id`) REFERENCES `event_prizes` (`id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 8. 포인트 사용/적립 테이블
CREATE TABLE IF NOT EXISTS `semo_point_history` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `user_id` INT NOT NULL,
    `amount` INT NOT NULL,
    `reason` VARCHAR(200) NULL,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
